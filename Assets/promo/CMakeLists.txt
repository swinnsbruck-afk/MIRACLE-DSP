cmake_minimum_required(VERSION 3.22)
project(PromoShots VERSION 0.1)
set(CMAKE_CXX_STANDARD 20)
add_subdirectory($ENV{HOME}/Developer/JUCE ${CMAKE_BINARY_DIR}/JUCE)
set(SRC "${CMAKE_CURRENT_SOURCE_DIR}/../../Source")
file(GLOB R "${CMAKE_CURRENT_SOURCE_DIR}/../../Assets/renders/*.png")
juce_add_binary_data(MiracleArtwork SOURCES ${R})
juce_add_console_app(PromoShots PRODUCT_NAME PromoShots)
target_sources(PromoShots PRIVATE promo_shots.cpp ${SRC}/MiracleEngine.cpp ${SRC}/SampleBank.cpp ${SRC}/PluginProcessor.cpp ${SRC}/PluginEditor.cpp)
target_include_directories(PromoShots PRIVATE ${SRC})
target_compile_definitions(PromoShots PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_MODAL_LOOPS_PERMITTED=1 JucePlugin_Name="MIRACLE DSP")
target_link_libraries(PromoShots PRIVATE juce::juce_audio_utils juce::juce_audio_formats MiracleArtwork juce::juce_recommended_config_flags)
